usuario base de datos postgres en cloud:
usuario: DB_USER=<insert-your-db-user>

superusuario base de datos postgres en cloud:
usuario: DJANGO_SUPERUSER=<insert-your-superuser-name>
password: DJANGO_SUPERUSER_PASSWORD=<insert-your-superuser-password>

psql -h DB_HOST=<insert-your-db-host> -U DB_USER=<insert-your-db-user> -d DB_NAME=<insert-your-db-name> -p 5432
